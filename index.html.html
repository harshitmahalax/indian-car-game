<script>

const car = document.getElementById("car");
const speedText = document.getElementById("speed");
const levelText = document.getElementById("level");
const carName = document.getElementById("carName");

const leftBtn = document.getElementById("left");
const rightBtn = document.getElementById("right");
const brakeBtn = document.getElementById("brake");
const hornBtn = document.getElementById("horn");

const menuBtn = document.getElementById("menuBtn");
const menu = document.getElementById("menu");

const cars = document.getElementById("cars");

let carPosition = 50;
let speed = 0;
let level = 1;
let police = false;
let led = false;
let alloy = false;

/* कार बाएँ */
function moveLeft() {

carPosition -= 5;

if (carPosition < 23)
carPosition = 23;

car.style.left = carPosition + "%";
}

/* कार दाएँ */
function moveRight() {

carPosition += 5;

if (carPosition > 77)
carPosition = 77;

car.style.left = carPosition + "%";
}

/* मोबाइल कंट्रोल */
leftBtn.addEventListener("touchstart", moveLeft);
rightBtn.addEventListener("touchstart", moveRight);

leftBtn.addEventListener("click", moveLeft);
rightBtn.addEventListener("click", moveRight);

/* कीबोर्ड */
document.addEventListener("keydown", function(e) {

if (e.key === "ArrowLeft")
moveLeft();

if (e.key === "ArrowRight")
moveRight();

});

/* स्पीड */
setInterval(function() {

if (speed < 120)
speed += 2;

speedText.textContent = speed;

if (speed > 80) {

level = Math.min(10, Math.floor(speed / 15));

levelText.textContent = level;
}

}, 500);

/* ब्रेक */
brakeBtn.addEventListener("click", function() {

speed -= 20;

if (speed < 0)
speed = 0;

speedText.textContent = speed;
});

/* हॉर्न */
hornBtn.addEventListener("click", function() {

hornBtn.textContent = "🔊";

setTimeout(function() {
hornBtn.textContent = "📣";
}, 500);

});

/* मेनू */
menuBtn.addEventListener("click", function() {

menu.style.display =
menu.style.display === "block"
? "none"
: "block";

});

/* कार बदलना */
cars.addEventListener("change", function() {

carName.textContent = cars.value;

});

/* पुलिस लाइट */
document.getElementById("police").addEventListener("click", function() {

police = !police;

car.style.boxShadow = police
? "0 0 30px red, 0 0 50px blue"
: "0 10px 25px #000";

});

/* एलईडी */
document.getElementById("lights").addEventListener("click", function() {

led = !led;

car.style.filter = led
? "brightness(1.5)"
: "brightness(1)";

});

/* अलॉय */
document.getElementById("alloy").addEventListener("click", function() {

alloy = !alloy;

car.style.border =
alloy
? "5px solid silver"
: "4px solid #222";

});

</script>
